<script>
    var canStealFocus = false;
    
    const additiveInput = document.getElementById("additiveTags");
    const requiredInput = document.getElementById("requiredTags");
    const subtractiveInput = document.getElementById("subtractiveTags");
    const orderInput = document.getElementById("pageOrder");
    const ratingComparisonInput = document.getElementById("ratingComparison");
    const ratingBaselineInput = document.getElementById("ratingBaseline");
    const votesComparisonInput = document.getElementById("votesComparison");
    const votesBaselineInput = document.getElementById("votesBaseline");
    const specificCreatorCheckbox = document.getElementById("specificCreatorCheckbox");
    const specificCreatorInput = document.getElementById("specificCreator");
    
    const comparisonOptions = ["any", ">", "<", "=", ">=", "<=", "<>"];
    function fillComparisonInput(input, linkedInput) {
    	for (const option of comparisonOptions) {
    		const node = document.createElement("option");
    		node.textContent = option;
    		node.setAttribute("value", option);
    		input.appendChild(node);
    	}
    	
    	input.addEventListener("change", function(e) {
    		linkedInput.style.display = (e.target.value == "any") ? "none" : "inline";
    		if (canStealFocus) linkedInput.focus();
    	}, { passive: true });
    }
    fillComparisonInput(votesComparisonInput, votesBaselineInput);
    fillComparisonInput(ratingComparisonInput, ratingBaselineInput);
    specificCreatorCheckbox.addEventListener("change", function(e) {
    	specificCreatorInput.style.display = (e.target.checked) ? "inline" : "none";
    	if (canStealFocus) specificCreatorInput.focus();
    });
    
    additiveInput.value = sessionStorage.getItem("additiveTags") || "";
    requiredInput.value = sessionStorage.getItem("requiredTags") || "";
    subtractiveInput.value = sessionStorage.getItem("subtractiveTags") || "";
    orderInput.value = sessionStorage.getItem("order") || "created_at desc";
    ratingComparisonInput.value = sessionStorage.getItem("ratingComparison") || "any";
    ratingComparisonInput.dispatchEvent(new Event("change"));
    ratingBaselineInput.value = sessionStorage.getItem("ratingBaseline") || "";
    votesComparisonInput.value = sessionStorage.getItem("votesComparison") || "any";
    votesComparisonInput.dispatchEvent(new Event("change"));
    votesBaselineInput.value = sessionStorage.getItem("votesBaseline") || "";
    specificCreatorCheckbox.checked = sessionStorage.getItem("specificCreatorCheckbox") == "true";
    specificCreatorCheckbox.dispatchEvent(new Event("change"));
    specificCreatorInput.value = sessionStorage.getItem("specificCreator") || "";
    canStealFocus = true;
    
    function getTagsFromInput(input, prefix) {
    	const str = input.value.trim();
    	if (str == "") return [];
    	
    	const array = str.split(/[ ,]+/);
    	for (let i = array.length - 1; i >= 0; i--) array[i] = prefix + array[i];
    	return array;
    }
    
    function goFunction() {
    	let voteBugPossible = orderInput.value.includes("votes");
    	let endStr = orderInput.value;
    	
    	const specificCreator = specificCreatorInput.value.trim();
    	if (specificCreatorCheckbox.checked && specificCreator) {
    		endStr += "/created_by/" + specificCreator;
    	}
    	
    	const ratingComparison = ratingComparisonInput.value;
    	const ratingBaseline = Number(ratingBaselineInput.value);
    	if (ratingComparison != "any" && ratingBaselineInput.value != "") {
    		if (isNaN(ratingBaseline) || ratingBaseline < 1 || ratingBaseline > 5) return alert("Error: Score must be a number from 1 to 5!");
    		endStr += "/rating/" + ratingComparison + (ratingBaseline * 10);
    	} else if (voteBugPossible) {
    		endStr += "/rating/>0";
    		voteBugPossible = false;
    	}
    	const votesComparison = votesComparisonInput.value;
    	const votesBaseline = parseInt(votesBaselineInput.value);
    	if (votesComparison != "any" && votesBaselineInput.value != "") {
    		if (isNaN(votesBaseline) || votesBaseline < 0 || votesBaseline != Number(votesBaselineInput.value)) return alert("Error: Votes must be a positive integer!");
    		endStr += "/votes/" + votesComparison + votesBaseline;
    	} else if (voteBugPossible) {
    		endStr += "/votes/<999999999";
    	}
    	
    	const additiveTags = getTagsFromInput(additiveInput, "");
    	const requiredTags = getTagsFromInput(requiredInput, "%2b");
    	const subtractiveTags = getTagsFromInput(subtractiveInput, "-");
    	let tagsStr = [...additiveTags, ...requiredTags, ...subtractiveTags].join();
    	if (tagsStr) tagsStr = "/tags/" + tagsStr;
    	
    	sessionStorage.setItem("additiveTags", additiveInput.value);
    	sessionStorage.setItem("requiredTags", requiredInput.value);
    	sessionStorage.setItem("subtractiveTags", subtractiveInput.value);
    	sessionStorage.setItem("order", orderInput.value);
    	sessionStorage.setItem("ratingComparison", ratingComparison);
    	sessionStorage.setItem("ratingBaseline", ratingBaselineInput.value);
    	sessionStorage.setItem("votesComparison", votesComparison);
    	sessionStorage.setItem("votesBaseline", votesBaselineInput.value);
    	sessionStorage.setItem("specificCreatorCheckbox", specificCreatorCheckbox.checked);
    	sessionStorage.setItem("specificCreator", specificCreatorInput.value);
    	
    	window.parent.location.replace("http://rpcauthority.wikidot.com/search:tags" + tagsStr + "/order/" + endStr + {$themeSuffix} + "#page-title");
    }
    
    document.getElementById("goButton").addEventListener("click", goFunction, { passive: true });
    document.getElementById("infoButton").addEventListener("click", function() {
    	document.getElementById("controlPanel").style.animation = "";
    	document.getElementById("controlPanel").style.visibility = "hidden";
    	document.getElementById("info").style.display = "block";
    }, { passive: true });
    document.getElementById("infoReturnButton").addEventListener("click", function() {
    	document.getElementById("controlPanel").style.animation = "fade 0.2s";
    	document.getElementById("controlPanel").style.visibility = "";
    	document.getElementById("info").style.display = "";
    }, { passive: true });
</script>
