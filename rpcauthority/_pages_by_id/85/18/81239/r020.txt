var saveState;

function focusfile(elem, bgrcolor, txtcolor) {
    var fileList = document.getElementsByClassName('file_selector');
    for (var i = 0; i < fileList.length; i++) {
        fileList[i].style.backgroundColor = "transparent";
        fileList[i].style.color = "white";
    }
    elem.style.backgroundColor = bgrcolor;
    elem.style.color = txtcolor;
}

function openfile(elem) {
    clearInterval(corruptInt);
    if (/\[CORRUPTED\]/g.test(elem.dataset.content)) {
        corruptLoop(elem);
    } else {
        document.getElementById('file_display').innerHTML = elem.dataset.content;
    }
}

function shortenNames() {
  var nameList = document.getElementsByClassName('file_name');
  for (var i = 0; i < nameList.length; i++) {
      if (nameList[i].innerHTML.length > 17) {
          nameList[i].innerHTML = nameList[i].textContent.substring(0,15) + "...";
      }
  }
}

function popup(color, title, content, hasinputbox, submitevent) {
    saveState = document.getElementById('layer').innerHTML;
    var popupCont = "<div class=\"divider popup\" id=\"popup\" style=\"border-color:" + color + "; color: " + color + "\"><strong>" + title + "</strong><br><br>" + content + "<br><br>";

    if (hasinputbox) {
        popupCont += "<form onsubmit=\"" + submitevent + "; closePopup()\"><input type=\"text\" id=\"popup_form\"><br><br><input type=\"submit\" value=\"OK\"></form>";
    } else {
        popupCont += "<button onclick=\"closePopup()\">OK</button>";
    }
    document.getElementById('layer').innerHTML = popupCont;
}

function closePopup() {
    document.getElementById('layer').innerHTML = saveState;
}

var itCount = 0;
var scrambleInt;

function scramble(elem) {
    var oldContent = elem.dataset.content;
    var scrambleContent = "";

    var re = /<.*?>|&.*?;/g;
    var stripList = oldContent.match(re);
    var strippedContent = oldContent.replace(re, "▔");

    var charList = ["!", "@", "#", "$", "%", "^", "&", "*", "(", ")"];
    for (var i = 0; i < strippedContent.length; i++) {
        var numRoll = Math.random() * 20;
        if (numRoll > itCount) {
          if (/▔/.test(strippedContent.charAt(i))) {
            scrambleContent += strippedContent.charAt(i);
          } else {
            scrambleContent += charList[Math.floor(Math.random() * charList.length)];
          }
        } else {
            scrambleContent += strippedContent.charAt(i);
        }
    }

    for (var i = 0; i < scrambleContent.length; i++) {
        var replaceRe = /▔/;
        scrambleContent = scrambleContent.replace(replaceRe, stripList[0]);
        stripList.splice(0, 1);
    }

    document.getElementById('file_display').innerHTML = scrambleContent;
    itCount++;

    clearInterval(scrambleInt);
    scrambleInt = setInterval(scramble, (100 * itCount / 5), elem);

    if (itCount >= 21) {
      clearInterval(scrambleInt);
      itCount = 0;
      document.getElementById('file_display').innerHTML = oldContent;
    }

    elem.dataset.scrambled = "false";
}



function scrambleLoop(elem) {
  if (elem.dataset.scrambled == "true") {
    scrambleInt = setInterval(scramble, 100, elem);
  } else {
    openfile(elem);
  }
}

var corruptInt;

function corruptLoop(elem) {
    corruptInt = setInterval(corrupt, 150, elem, 6);
}

function corrupt(elem, corrLength) {
    var corruptCharList = ["▖", "▗", "▘", "▙", "▚", "▛", "▜", "▝", "▞"];
    var corruptStringMain = elem.dataset.content;
    var corruptCount = corruptStringMain.split("[CORRUPTED]").length;

    for (var a = 0; a < corruptCount; a++) {
        var corruptString = "";
        for (var b = 0; b < corrLength; b++) {
            corruptString += corruptCharList[Math.floor(Math.random() * corruptCharList.length)];
        }
        corruptStringMain = corruptStringMain.replace(/\[CORRUPTED\]/, "<span style='color:#BC111B'>" + corruptString + "</span>");
    }

    document.getElementById('file_display').innerHTML = corruptStringMain;
}
