**Пограничье** — Мир ближней ветви, открытый в 1972 году. Вследствие своей близости по древу слабо отличается от нулевого. Главной характерной особенностью Мира можно назвать невозможность обратного перемещения живой материи. Любое живое существо, перемещенное из Мира 2 в любой другой неизбежно погибает.

 Мир находится на орбите двойной звездной системы, состоящей из белого и красного карликов, Гелиоса и Санга Лепоро, соответственно. Помимо единственной обитаемой планеты, названной **Элрондом**, на орбите звездной пары находятся раскаленный "горячий Юпитер" Касиус, пустынный, изрезанный каньонами Кайрис и покрытая облаками Каллеопа.
Климат Элронда жаркий. Малый эксцентриситет орбиты обеспечивает ровную смену времен года, которые сменяются каждые 17 месяцев. Осень и весна выражены слабо. Долгое лето сопровождается высокими температурами, на экваторе достигающими 70°C. В зимнее время года снег выпадает лишь на полярных областях. Сутки продолжаются 30 Земных часов. Вследствие подобных отличий, была разработана новая система измерения времени. За стандартную единицу был принят Земной час, соответственно на местных часах оных насчитывается 30. Год делится на шесть месяцев, в каждом из которых числится по 170 дней. Для удобства месяцы были названы соответственно земным: декабрь, январь, февраль, летние июнь, июль и август.

В настоящее время доподлинно известны лишь три материка. **Эйрис** является самым северным из известных, в его северной части пролегает экватор. Это наиболее жаркий континент, жизнь на котором возможна лишь на южных его берегах. Центральная часть находится на стыке крупнейших литосферных плит и представляет собой непроходимый горный массив, в самой высокой своей части достигающий 14 километров. Южная часть материка покрыта гигантской раскаленной пустыней. Именно эта пустыня, именуемая "Черной" и является основным месторождением нефти, выходящей на поверхность в виде луж или целых озер. На территории этой пустыни, а также в горах к северу обитают многочисленные дикие племена людей, попавших на Пограничье еще до начала колонизации. Впрочем, юго-восточная часть материка, благодаря холодному течению покрыта густым лиственным лесом и вполне пригодна для комфортного проживания.

Центральный континент, **Аркан** имеет вулканическую природу. Судя по всему, он возник более двух млрд. лет назад вследствие извержения супервулкана, разделившего единый ранее континент. Вулкан считается давно потухшим, однако продукты его деятельности обеспечили прекрасную базу для растительной жизни. Большая часть континента покрыта непроходимыми жаркими джунглями вечнозеленых растений. Сложности со свободным для строительства местом вынуждают местных колонистов осваивать строительство на деревьях. Именно здесь можно встретить Грузотигр — ловких и сильных травоядных животных, которые прекрасно приручаются и экспортируются по всей планете в качестве покладистых и выносливых ездовых или вьючных животных.

Южный континент — **Титус**, в данный момент является наиболее заселенным. Северная часть материка покрыта жаркими саваннами и прериями со скудной растительностью, во многих местах сменяющимися пустынями. Северо-западная оконечность омывающаяся холодным течением менее засушлива и пригодна для комфортного проживания. Юг материка покрыт жаркими непроходимыми болотами и в данный момент изучен слабо. В середине декабря Слоновый пролив мельчает, на его месте образуется перешеек Святой Линды, который служит удобным, но не слишком безопасным торговым маршрутом. Именно в это время открывается Тигриный торговый путь, по которому приходят в движение множество караванов.

<script>
"use strict";
/* Hours in days, months and years in Frontier */
const fdayhours = 30;
const fmonthhours = 170*fdayhours;
const fyearhours = 6*fmonthhours;
const origin = new Date("1972-03-01");
const monthnames = ["января", "февраля", "марта", "апреля", "мая", "июня", "июля", "августа", "сентября", "октября", "ноября", "декабря"];
const fmonthnames = ["января", "февраля", "июня", "июля", "августа", "декабря"];

var ea;
{
    var dayhours = 24;
    var monthhours = dayhours*30;
    var yearhours = dayhours*365;
    ea = [dayhours, monthhours, yearhours];
}
var fr = [fdayhours, fmonthhours, fyearhours];

function time(hours, to) {
  var divrem = (x, y) => [Math.floor(x / y), x % y];
  var [ryears, hours] = divrem(hours, to[2], 1);
  var [rmonths, hours] = divrem(hours, to[1], 1);
  var rdays = Math.floor(hours/to[0]*100)/100;
  return [rdays, rmonths, ryears];
}

function initTime() {
  [["day", "Дней"], ["month", "Месяцев"], ["year", "Лет"]].forEach(([a, b]) => {
    document.getElementById(`date-${a}-label`).innerText = b;
  });
  fieldDay.max = "";
  var monthWrap = document.getElementById("date-month");
  monthWrap.innerHTML = "";
  fieldMonth = document.createElement("input");
  fieldMonth.type = "number";
  fieldMonth.addEventListener("change", recalc);
  monthWrap.appendChild(fieldMonth);
}

// It's better to join this in a single function, but we both know that I won't.
function initDate(mx) {
  [["day", "Число"], ["month", "Месяц"], ["year", "Год"]].forEach(([a, b]) => {
    document.getElementById(`date-${a}-label`).innerText = b;
  });
  fieldDay.max = mx;
  var monthWrap = document.getElementById("date-month");
  monthWrap.innerHTML = "";
  fieldMonth = document.createElement("select");
  var usableMonths;
  if(fromFrontier) {
    usableMonths = fmonthnames;
  } else {
    usableMonths = monthnames;
  }
  usableMonths.forEach(x => {
    var opt = document.createElement("option");
    opt.innerText = x;
    fieldMonth.appendChild(opt);
  });
  fieldMonth.addEventListener("change", recalc);
  monthWrap.appendChild(fieldMonth);
}

var fieldDay;
var fieldMonth;
var fieldYear;
var fieldOut;
var fieldType;

var translateTime = true;
var fromFrontier = true;

function recalc() {
  if(translateTime) {
    var from, to;
    if(fromFrontier) {
      to = ea;
      from = fr;
    } else {
      to = fr;
      from = ea;
    }
    var hours = (fieldDay.valueAsNumber || 0)*from[0] + (fieldMonth.valueAsNumber || 0)*from[1] + (fieldYear.valueAsNumber || 0)*from[2];
    var [days, months, years] = time(hours, to);
    fieldOut.innerText = `${days} дней, ${months} месяцев, ${years} лет`;
  } else {
    /* Translate date */
    var day = fieldDay.valueAsNumber || 1;
    var month = fieldMonth.selectedIndex + 1;
    var year = fieldYear.valueAsNumber || 1;
    if(fromFrontier) {
      var hours = (day-1)*fdayhours + (month-1)* fmonthhours + (year-1)* fyearhours;
      var out = new Date(origin);
      out.setHours(origin.getHours()+hours);
      fieldOut.innerText = `${out.getDate()} ${monthnames[out.getMonth()]} ${out.getFullYear()} года`;
    } else {
      var hours = (new Date(`${year}-${month}-${day}`)-origin)/1000/60/60;
      if(!isNaN(hours)) {
        var [day, month, year] = time(hours, fr);
        day = Math.floor(day);
        fieldOut.innerText = `${day+1} ${fmonthnames[month]} ${year+1} года`;
      } else {
        fieldOut.innerText = "";
      }
    }
  }
}

function reselect() {
  var val = fieldType.value;
  // Maybe someone could use object here instead of these else ifs.
  if(val == "tf->te") {
    fromFrontier = true;
    translateTime = true;
    initTime();
  } else if (val == "te->tf") {
    fromFrontier = false;
    translateTime = true;
    initTime();
  } else if (val == "df->de") {
    fromFrontier = true;
    translateTime = false;
    initDate(170);
  } else if (val == "de->df") {
    fromFrontier = false;
    translateTime = false;
    initDate(31);
  }
  recalc();
}

window.onload = () => {
  fieldType = document.getElementById("conv-type");
  fieldType.addEventListener("change", reselect);
  fieldOut = document.getElementById("out");
  fieldDay = document.getElementById("date-day");
  fieldYear = document.getElementById("date-year");
  [fieldDay, fieldYear].forEach(x => x.addEventListener("change", recalc));
  initTime();
};
</script>
