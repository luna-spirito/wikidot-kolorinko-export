> [[include include:object
>    cat =Категория 1
> | cat2 =Категория 2
> | dan =Уровень угрозы 1
> | dan2 =Уровень угрозы 2|
> | iso =Уровень изоляции 1|
> | iso2 =Уровень изоляции 2|
> | s1 = Эти три переменные содержат наименования переменных, которые нужно зачеркнуть
> | s2 = Возможные значения:
> | s2 =  cat/cat2/dan/dan2/iso/iso2
> ]]

> [[include include:object
> cat=Физический |
> cat2=Трансреальный|

> iso2=3 |
> s1 = cat |
> s2 = iso
> ]]
