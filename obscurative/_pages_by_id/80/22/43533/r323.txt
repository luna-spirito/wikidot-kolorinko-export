let counter = -1;
let descId = document.getElementById("desc");
let helperId = document.getElementById("helper");

var hold = {
    " ": 150,
    "\n": 250,
    ".": 225,
    ",": 175
};
var d_hold = 60;

function finitaLaCommedia() {  // But in case of JS this is some kind of a tragedy
  document.getElementById("btns").style.display = "none";
}

function updateText(forDesc, forHelper) {
  descId.innerHTML = "";
  let cchr = 0;
  let bcounter = counter;
  function help() {
    if(forHelper != undefined) {
    setTimeout(function() {
      helperId.innerHTML = forHelper;
      helperId.style.opacity = "1";
      }, 1000);
    }
  }
  function type_in() {
    if(bcounter != counter)
      return;
    let chr = forDesc[cchr];
    descId.innerHTML += chr;
    if(++cchr != forDesc.length) {
      var delay = hold[chr];
      if(delay == undefined)
        delay = d_hold;
      setTimeout(type_in, type_in);
    } else
      help();
  }
  type_in();
  helperId.innerHTML = "";
  helperId.style.opacity = "0";
}

function yes() {
  if(++counter == desc.length-1)
    finitaLaCommedia();
  updateText(desc[counter], helper[counter]);
}

function no() {
  updateText(decline, "");
  finitaLaCommedia();
}
