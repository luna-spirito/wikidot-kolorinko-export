<script>
"use strict";
/* Hours in days, months and years in Frontier */
const fdayhours = 30;
const fmonthhours = 170*fdayhours;
const fyearhours = 6*fmonthhours;
const origin = new Date("1972-03-01");
const monthnames = ["января", "февраля", "марта", "апреля", "мая", "июня", "июля", "августа", "сентября", "октября", "ноября", "декабря"];
const fmonthnames = ["января", "февраля", "июня", "июля", "августа", "декабря"];

var ea;
{
    var dayhours = 24;
    var monthhours = dayhours*30;
    var yearhours = dayhours*365;
    ea = [dayhours, monthhours, yearhours];
}
var fr = [fdayhours, fmonthhours, fyearhours];

function time(hours, to) {
  var divrem = (x, y) => [Math.floor(x / y), x % y];
  var [ryears, hours] = divrem(hours, to[2], 1);
  var [rmonths, hours] = divrem(hours, to[1], 1);
  var rdays = Math.floor(hours/to[0]*100)/100;
  return [rdays, rmonths, ryears];
}

function initTime() {
  [["day", "Дней"], ["month", "Месяцев"], ["year", "Лет"]].forEach(([a, b]) => {
    document.getElementById(`date-${a}-label`).innerText = b;
  });
  fieldDay.max = "";
  var monthWrap = document.getElementById("date-month");
  monthWrap.innerHTML = "";
  fieldMonth = document.createElement("input");
  fieldMonth.type = "number";
  fieldMonth.addEventListener("change", recalc);
  monthWrap.appendChild(fieldMonth);
}

// It's better to join this in a single function, but we both know that I won't.
function initDate(mx) {
  [["day", "Число"], ["month", "Месяц"], ["year", "Год"]].forEach(([a, b]) => {
    document.getElementById(`date-${a}-label`).innerText = b;
  });
  fieldDay.max = mx;
  var monthWrap = document.getElementById("date-month");
  monthWrap.innerHTML = "";
  fieldMonth = document.createElement("select");
  var usableMonths;
  if(fromFrontier) {
    usableMonths = fmonthnames;
  } else {
    usableMonths = monthnames;
  }
  usableMonths.forEach(x => {
    var opt = document.createElement("option");
    opt.innerText = x;
    fieldMonth.appendChild(opt);
  });
  fieldMonth.addEventListener("change", recalc);
  monthWrap.appendChild(fieldMonth);
}

var fieldDay;
var fieldMonth;
var fieldYear;
var fieldOut;
var fieldType;

var translateTime = true;
var fromFrontier = true;

function recalc() {
  if(translateTime) {
    var from, to;
    if(fromFrontier) {
      to = ea;
      from = fr;
    } else {
      to = fr;
      from = ea;
    }
    var hours = (fieldDay.valueAsNumber || 0)*from[0] + (fieldMonth.valueAsNumber || 0)*from[1] + (fieldYear.valueAsNumber || 0)*from[2];
    var [days, months, years] = time(hours, to);
    fieldOut.innerText = `${days} дней, ${months} месяцев, ${years} лет`;
  } else {
    /* Translate date */
    var day = fieldDay.valueAsNumber || 1;
    var month = fieldMonth.selectedIndex + 1;
    var year = fieldYear.valueAsNumber || 1;
    if(fromFrontier) {
      var hours = (day-1)*fdayhours + (month-1)* fmonthhours + (year-1)* fyearhours;
      var out = new Date(origin);
      out.setHours(origin.getHours()+hours);
      fieldOut.innerText = `${out.getDate()} ${monthnames[out.getMonth()]} ${out.getFullYear()} года`;
    } else {
      var hours = (new Date(`${year}-${month}-${day}`)-origin)/1000/60/60;
      if(!isNaN(hours)) {
        var [day, month, year] = time(hours, fr);
        console.log(month);
        fieldOut.innerText = `${day+1} ${fmonthnames[month+1]} ${year+1} года`;
      } else {
        fieldOut.innerText = "";
      }
    }
  }
}

function reselect() {
  var val = fieldType.value;
  // Maybe someone could use object here instead of these else ifs.
  if(val == "tf->te") {
    fromFrontier = true;
    translateTime = true;
    initTime();
  } else if (val == "te->tf") {
    fromFrontier = false;
    translateTime = true;
    initTime();
  } else if (val == "df->de") {
    fromFrontier = true;
    translateTime = false;
    initDate(170);
  } else if (val == "de->df") {
    fromFrontier = false;
    translateTime = false;
    initDate(31);
  }
  recalc();
}

window.onload = () => {
  fieldType = document.getElementById("conv-type");
  fieldType.addEventListener("change", reselect);
  fieldOut = document.getElementById("out");
  fieldDay = document.getElementById("date-day");
  fieldYear = document.getElementById("date-year");
  [fieldDay, fieldYear].forEach(x => x.addEventListener("change", recalc));
  initTime();
};
</script>
