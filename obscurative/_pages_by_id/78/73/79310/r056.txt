<script>
var text = document.getElementById("text");
var buttons = document.getElementById("buttons");
var url = "http://university-of-pandora.wdfiles.com/local--files/draft%3Agame/";
var location = {};
function log(msg) {
  text.innerHTML += msg+"\n";
}
function talk(dialog) {
  let speaker = "";
  alert("talking: "+dialog);
  if(typeof dialog === 'string')
    dialog = [dialog];
  for(let i = 0; i < dialog.length; i++) {
    let val = dialog[i];
    if(typeof val === 'object') {
      speaker = val[0];
      val = val[1];
    }
    let msg = "";
    if(speaker != "")
      msg = speaker+": ";
    log(msg+val);
    alert("works!");
  }
}
function load(loc) {
  let xhr;
  let timeout;
  function post() {
    xhr = new XMLHttpRequest()
    xhr.open("GET", url+loc+".json");
    xhr.addEventListener("loadend", end);
    xhr.send();
    timeout = setTimeout(abort, 1500);
  }
  function end() {
    if(xhr.status == 200) {
      location = JSON.parse(xhr.responseText);
      talk(location.description);
    } else {
      clearTimeout(timeout);
      setTimeout(post, 500);
    }
  }
  function abort() {
    xhr.abort();
    setTimeout(post, 500);
  }
  post();
}

load("start");
</script>
