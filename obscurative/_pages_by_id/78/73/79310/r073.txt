<script>
var text = document.getElementById("text");
var buttons = document.getElementById("buttons");
var url = "http://university-of-pandora.wdfiles.com/local--files/draft%3Agame/";
var loc = {"description": 3};
function log(msg) {
  text.innerHTML += msg+"\n";
}
function talk(branch) {
  let speaker = "";
  dialog = branch.text;
  if(typeof dialog === 'string')
    dialog = [dialog];
  for(let i = 0; i < dialog.length; i++) {
    let val = dialog[i];
    if(typeof val === 'object') {
      speaker = val[0];
      val = val[1];
    }
    let msg = "";
    if(speaker != "")
      msg = speaker+": ";
    log(msg+val);
    addButtons();
  }
}
function load(id) {
  let xhr;
  let timeout;
  function post() {
    xhr = new XMLHttpRequest()
    xhr.open("GET", url+id+".json");
    xhr.timeout = 3000;
    xhr.ontimeout = function() {alert("timeout!"); setTimeout(post, 0);};
    xhr.addEventListener("loadend", end);
    xhr.responseType="json";
    xhr.send();
  }
  function end() {
    clearTimeout(timeout);
    if(xhr.status == 200) {
      loc = xhr.response;
      talk(loc);
    } else {
      setTimeout(post, 0);
    }
  }
  post();
}

function addButtons(options) {
  if(!Array.isArray(options))
    options = [options];
  for(let i = 0; i < options.length; i++) {
    var button = document.createElement('button');
    button.addEventListener("click", function() {alert("Hi!")});
    buttons.append(button);
  }
}

load("start");
</script>
