<script>
var text = document.getElementById("text");
var buttons = document.getElementById("buttons");
var url = "http://university-of-pandora.wdfiles.com/local--files/draft%3Agame/";
var loc = {};
function _log(elem) {
  let scroll = text.scrollTop == text.scrollTopMax;
  let available = true;
  if(!scroll && typeof text.scrollTo === 'undefined') {
    available = false;
    let temp = text.scrollTop;
    text.scrollTop += 1;
    if(temp == text.scrollTop)
      scroll = true;
    text.scrollTop -= 1;
  }
  text.appendChild(elem);
  if(scroll)
    if(available)
      text.scrollTo({
        top: text.scrollTopMax,
        behavior: "smooth"
      });
    else
      text.scrollTop = text.scrollHeight;
  setTimeout(function() {elem.style.opacity = 1;}, 50);
}
function log(msg) {
  let elem = document.createElement("p");
  elem.innerHTML = msg;
  _log(elem);
}
function talk(branch) {
  let speaker = "";
  dialog = branch.text;
  if(typeof dialog === 'string')
    dialog = [dialog];
  let i = 0;
  function repeat() {
    let val = dialog[i];
    if(typeof val === 'object') {
      speaker = val[0];
      val = val[1];
    }
    let msg = "";
    if(speaker != "")
      msg = speaker+": ";
    msg = msg+val;
    log(msg);
    i++;
    if(i>=dialog.length) {
      addButtons(branch.responses);
    } else
      setTimeout(repeat, val.length*8);
  }
  setTimeout(repeat, 100);
}
function load(id) {
  let xhr;
  let timeout;
  function post() {
    xhr = new XMLHttpRequest()
    xhr.open("GET", url+id+".json");
    xhr.timeout = 3000;
    xhr.ontimeout = function() {alert("timeout!"); setTimeout(post, 0);};
    xhr.addEventListener("loadend", end);
    ///xhr.responseType="json";
    xhr.send();
  }
  function end() {
    clearTimeout(timeout);
    if(xhr.status == 200) {
      loc = JSON.parse(xhr.responseText);
      talk(loc);
    } else {
      setTimeout(post, 0);
    }
  }
  post();
}

function addButtons(options) {
  if(typeof options === "undefined")
    return;
  if(!Array.isArray(options))
    options = [options];
  for(let i = 0; i < options.length; i++) {
    let option = options[i];
    let button = document.createElement('button');
    button.addEventListener("click", function() {select(option);});
    button.innerHTML = option.response;
    let li = document.createElement("li");
    li.appendChild(button);
    buttons.appendChild(li);
    setTimeout(function() {li.style.opacity = 1;}, 50);
  }
}

function select(response) {
  buttons.innerHTML = "";
  let elem = document.createElement("li");
  elem.innerHTML = response.response;
  _log(elem);
  setTimeout(function() {talk(response);}, 400);
}

load("start");
</script>
